-- Kwargs UI Library v3 · Example
local Lib = loadstring(game:HttpGet("https://raw.githubusercontent.com/anhyeeuemnhieulam-ai/makingkiba/refs/heads/main/source%20(1).lua"))()

local UI = Lib:Create{
	Name = "Kwargs Hub",
	Theme = "Dark",                       -- Dark, Light, Discord, Red, Ocean, Midnight, Rose, or your own table
	Size = UDim2.new(0, 580, 0, 400),     -- default
	ToggleKey = Enum.KeyCode.RightControl, -- show / hide the window
	Animated = true                        -- false = no looping effects (low-end devices)
}

----------------------------------------------------------------------
-- Main tab
----------------------------------------------------------------------
local Main = UI:Tab{ Name = "Main" }
local Controls = Main:Divider{ Name = "Controls" }

Controls:Button{
	Name = "Say hello",
	Description = "Shows a notification.",
	Callback = function()
		UI:Notify{ Title = "Hello!", Content = "Kwargs UI v2 is running.", Type = "Success", Duration = 4 }
	end
}

Controls:Toggle{
	Name = "Toggle (style 1)",
	Description = "Every element supports Name, Description and Callback.",
	Flag = "toggle1",                     -- Flag = saved in configs
	Callback = function(State) print("Toggle:", State) end
}

Controls:Toggle{
	Name = "Toggle (style 2)",
	Style = 2,
	State = true,
	Flag = "toggle2"
}

Controls:Slider{
	Name = "Speed",
	Min = 0, Max = 100, Default = 16, Increment = 1, Suffix = " st/s",
	Flag = "speed",
	Callback = function(Value) print("Speed:", Value) end
}

Controls:Dropdown{
	Name = "Player list",
	Options = { "Player1", "Player2", "Player3", "Player4", "Player5" },
	Flag = "player",
	Callback = function(Value) print(Value) end
}

Controls:Dropdown{
	Name = "Multi select",
	Options = { "Apple", "Banana", "Cherry", "Mango" },
	Multi = true,
	Default = { "Apple" },
	Flag = "fruits",
	Callback = function(Values) print(table.concat(Values, ", ")) end
}

Controls:SearchDropdown{
	Name = "Teleports",
	Options = { "Pleasant Park", "Loot Lake", "Tomato Town", "Wailing Woods", "Anarchy Acres", "Retail Row" },
	Callback = function(Value) print(Value) end
}

Controls:ColorPicker{
	Name = "Accent color",
	Default = Color3.fromRGB(0, 255, 255),
	Flag = "color",
	Callback = function(Value) print(Value) end
}

Controls:Box{
	Name = "Display name",
	Placeholder = "Type here...",
	ClearText = true,                     -- clear text when focused
	Flag = "name",
	Callback = function(Text) print(Text) end
}

Controls:Keybind{
	Name = "Action key",
	Default = Enum.KeyCode.E,
	Flag = "bind",
	Callback = function(Key) print("Pressed", Key.Name) end
}

Controls:Label{ Text = "Tip: press <b>Backspace</b> or <b>Esc</b> while binding a key to clear it." }

----------------------------------------------------------------------
-- Settings tab
----------------------------------------------------------------------
local Settings = UI:Tab{ Name = "Settings" }

local Appearance = Settings:Divider{ Name = "Appearance" }
Appearance:Dropdown{
	Name = "Theme",
	Options = { "Dark", "Light", "Discord", "Red", "Ocean", "Midnight", "Rose" },
	Default = "Dark",
	Callback = function(Theme) UI:SetTheme(Theme) end -- switches live, no reload
}

local Config = Settings:Divider{ Name = "Config" }
Config:Button{
	Name = "Save config",
	Callback = function()
		local ok, err = UI:SaveConfig("default")
		UI:Notify{ Title = ok and "Config saved" or "Save failed", Content = err, Type = ok and "Success" or "Error" }
	end
}
Config:Button{
	Name = "Load config",
	Callback = function()
		local ok, err = UI:LoadConfig("default")
		UI:Notify{ Title = ok and "Config loaded" or "Load failed", Content = err, Type = ok and "Success" or "Error" }
	end
}

local Quit = Settings:Divider{ Name = "Quit" }
Quit:Button{
	Name = "Close the UI",
	Callback = function()
		UI:Quit{ Message = "Goodbye!", Length = 1 }
	end
}
